### Step 4 — Shadow-load migration helper

```cpp
// v5.15.5.E.0 — Shadow-load migration: read legacy raw-fwrite bytes,
// populate runtime ConfidenceScorer via field-by-field copy. Composite-
// mode fields are RE-INIT from cfg + Init defaults (re-warm correct;
// stale wall-clock/EWMA state would be wrong). Per
// shadow-load-state-transition-pattern.md.
//
// Returns 0 on success; -1 on fread failure.
static inline int ConfidenceScorer_ShadowLoadLegacyV1(ConfidenceScorer* cs, FILE* f) {
    ConfidenceScorerLegacyV1 wire;
    if (fread(&wire, sizeof(wire), 1, f) != 1) return -1;

    // Re-init runtime struct to defaults (clean slate; composite fields
    // get default-on-disabled state per Init).
    ConfidenceScorer_Init(cs, /*window=*/wire.ic_window, /*tau=*/wire.freshness_tau);

    // Copy persisted IC + RMSE history from wire to runtime.
    memcpy(cs->ic.predictions, wire.predictions, sizeof(wire.predictions));
    memcpy(cs->ic.actuals,     wire.actuals,     sizeof(wire.actuals));
    cs->ic.count               = wire.ic_count;
    cs->ic.head                = wire.ic_head;
    memcpy(cs->rmse.squared_errors, wire.rmse_squared_errors, sizeof(wire.rmse_squared_errors));
    cs->rmse.count             = wire.rmse_count;
    cs->rmse.head              = wire.rmse_head;
    cs->last_confidence        = wire.last_confidence;
    // Composite-mode fields RE-INIT (per Init); wire.freshness_*, wire.capacity_*,
    // wire.rmse_baseline are intentionally DROPPED — wall-clock + EWMA stale on
    // reload; cfg re-bind handles them at boot via ConfidenceScorer_BindCompositeCfg.

    return 0;
}
```

### Step 2 — Maintain sum_squared_errors at Push

```cpp
static inline void RollingRMSE_Push(RollingRMSE* r, double prediction, double actual) {
    double err = prediction - actual;
    double new_se = err * err;
    int idx = r->window.head % r->window.window;
    if (r->window.count >= r->window.window) {
        // Evict oldest before write
        r->sum_squared_errors -= r->window.samples[idx];
    }
    r->window.samples[idx] = new_se;
    r->sum_squared_errors += new_se;
    r->window.head++;
    if (r->window.count < r->window.window) r->window.count++;
}
```

### Step 3 — O(1) RollingRMSE_Compute

```cpp
static inline double RollingRMSE_Compute(const RollingRMSE* r) {
    if (r->window.count < 2) return 1.0;
    return sqrt(r->sum_squared_errors / (double)r->window.count);
}
```
