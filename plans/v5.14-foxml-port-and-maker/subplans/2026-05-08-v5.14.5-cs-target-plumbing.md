// SIGNATURE: matches canonical LabelFn typedef at LabelFunctions.hpp:284
// `(ticks, tick_idx, total_ticks, sample_price, tp_pct, sl_pct, extra_param)`.
// CS targets ignore tp_pct/sl_pct (no barriers); use extra_param =
// forward_ticks for horizon. Trace-deps fix 2026-05-08.

// CS Percentile Rank: per-timestamp rank / (N+1).
// Single-symbol degenerate case: returns raw return (CS rank
// aggregation happens at BacktestResults level when multi-symbol
// architecture lands in v5.16+).
static inline float Label_CSPercentileRank(
    const HistoricalTick *ticks, int tick_idx, int total_ticks,
    double sample_price, double /*tp_pct*/, double /*sl_pct*/,
    int forward_ticks) {
    if (tick_idx + forward_ticks >= total_ticks) return NAN;
    if (sample_price <= 0.0) return NAN;
    double future_return = (ticks[tick_idx + forward_ticks].price -
                             sample_price) / sample_price;
    return (float)future_return;
}

// CS Robust Z-Score: (r - median) / (1.4826 × MAD).
// Single-symbol: returns raw return (CS-bucketed normalization
// applied at BacktestResults aggregation step).
static inline float Label_CSZScoreRobust(
    const HistoricalTick *ticks, int tick_idx, int total_ticks,
    double sample_price, double /*tp_pct*/, double /*sl_pct*/,
    int forward_ticks) {
    if (tick_idx + forward_ticks >= total_ticks) return NAN;
    if (sample_price <= 0.0) return NAN;
    double future_return = (ticks[tick_idx + forward_ticks].price -
                             sample_price) / sample_price;
    return (float)future_return;
}

// CS Vol-Scaled Demeaned: (r/vol) - mean_cs(r/vol).
// Vol estimate sourced from rolling stats at BacktestResults level
// (NOT from cfg — cfg threading would require dispatcher refactor;
// the LabelFn signature is locked). For single-symbol: returns raw
// return; vol-scaling happens at aggregation.
static inline float Label_CSVolScaledDemeaned(
    const HistoricalTick *ticks, int tick_idx, int total_ticks,
    double sample_price, double /*tp_pct*/, double /*sl_pct*/,
    int forward_ticks) {
    if (tick_idx + forward_ticks >= total_ticks) return NAN;
    if (sample_price <= 0.0) return NAN;
    double future_return = (ticks[tick_idx + forward_ticks].price -
                             sample_price) / sample_price;
    return (float)future_return;
}

## Step 5 — Fractional differentiation (v5.14.5.C; RESTORED 2026-05-08)

**DESIGN DISCOVERY (2026-05-08, post-deferral):** The original deferral
rationale was wrong. `RollingStats<F, W>` ALREADY contains the raw
ring buffer needed for frac diff:

```cpp
// From ML_Headers/RollingStats.hpp:89-91
FPN<F> price_buf[W];   // raw price ring
FPN<F> volume_buf[W];  // raw volume ring
```

Plus `head` (write position, line 77), `count` (samples buffered,
line 78). `FeatureComputeCtx<F>::short_rolling` is a pointer to a
`RollingStats<F, 128>` — so `ctx->short_rolling->price_buf[]` IS the
raw price history, accessible from feature compute fns with zero
new infrastructure.

The deferral was a Class 17 bug (architectural deferral made without
grepping adjacent struct fields) — caught + reverted same day. See
RECURRING_BUG_PATTERNS.md Class 17 for full breakdown.

**Pre-existing work audit:**
- `RollingStats<F, W>::price_buf[W]` exists at `ML_Headers/RollingStats.hpp:90`
- `RollingStats<F, W>::volume_buf[W]` exists at `ML_Headers/RollingStats.hpp:91`
- `RollingStats<F, W>::head` (write position) at `:77`
- `RollingStats<F, W>::count` (warmup state) at `:78`
- `FeatureComputeCtx<F>::short_rolling` pointer at `FeatureRegistry.hpp:76`
  (W=128 → 128 lags available)
- W=128 sufficient: |C(0.5, 50)| < 1e-6, frac diff truncates well
  before W

**TRULY NEW:**
- 3 new FOREACH_FEATURE rows:
  - `ML_Compute_FracDiffPrice_d04` — d=0.4 frac diff of price
  - `ML_Compute_FracDiffPrice_d05` — d=0.5 frac diff of price
  - `ML_Compute_FracDiffPrice_d06` — d=0.6 frac diff of price
- 3 inline functions reading `ctx->short_rolling->price_buf[]`
- 3 precomputed coefficient tables (`constexpr` arrays of FPN<F>
  binomial coefficients C(d, k) for k=0..K-1, K=50)

**Math:**
```
Δ^d x_t = Σ_{k=0..K-1} (-1)^k * C(d, k) * x_{t-k}

C(d, k) = Γ(d+1) / (Γ(k+1) * Γ(d-k+1))
        = d * (d-1) * (d-2) * ... * (d-k+1) / k!
```

For d ∈ {0.4, 0.5, 0.6}, K=50 captures > 99.999% of the infinite-sum
weight. Coefficients are float-constant so precomputed at compile time.

**Implementation pattern (ML_Compute_FracDiffPrice_d05):**

```cpp
// Precomputed coefficients C(0.5, k) for k=0..49 (constexpr table).
// Storage: 50 × FPN<64>=24B = 1200B = 19 cache lines (read-only,
// cold-on-arrival; once-per-feature-pack so no L1 pressure).
inline constexpr FPN<F> kFracDiff_d05_Coeffs[50] = { /* table */ };

template <unsigned F>
inline FPN<F> ML_Compute_FracDiffPrice_d05(const FeatureComputeCtx<F>* ctx) {
    if (!ctx || !ctx->short_rolling) return FPN_Zero<F>();
    const RollingStats<F, 128>* rs = ctx->short_rolling;

    // Cold-start: not enough history → return zero.
    if (rs->count < 50) return FPN_Zero<F>();

    // Walk K=50 most-recent prices: index = (head - 1 - k) & (W-1)
    // FPN math (no float). Branchless wrap (W=128 power-of-2).
    FPN<F> sum = FPN_Zero<F>();
    constexpr int W = 128;
    int idx = (rs->head - 1) & (W - 1);
    for (int k = 0; k < 50; k++) {
        // Sign alternates: +, -, +, -, ... → -(k & 1) mask trick
        FPN<F> term = FPN_Mul<F>(kFracDiff_d05_Coeffs[k], rs->price_buf[idx]);
        sum = ((k & 1) == 0) ? FPN_Add<F>(sum, term) : FPN_Sub<F>(sum, term);
        idx = (idx - 1) & (W - 1);
    }
    return sum;
}
```

(Sister functions `_d04` + `_d06` identical shape with their own
coefficient tables.)

**LATENCY:** O(K=50) per feature; FPN_Mul + FPN_Add per iter ≈ 5-8ns
each; ~300-400ns per feature; ~1μs total for 3 features. SLOW-PATH;
within 100μs budget. Coefficient table is constexpr → compiled into
.rodata, no init cost. Branchless wrap. No allocations.

**Train-serve parity:** SAME function reads SAME `price_buf` ring in
both live + backtest paths. Live: producer fans real ticks → slow-path
calls `RollingStats_Push` populating ring → ML compute reads ring at
slow-path cadence. Backtest: same Push mechanism via
`RollingStats_Push(rs, hist_tick.price, hist_tick.volume)` in the
backtest driver. Bytewise identical to live.

**Tests (~7):**
- Steady-state: known input ticks → frac diff matches reference
  Python implementation to within FPN<64> precision (1/2^64)
- Cold-start (count < K=50): returns FPN_Zero
- Trend: monotonic ramp → frac diff captures fractional rate
- Stationary noise: frac diff ~zero mean (regression sanity)
- Coefficient table sanity: sum of |C(0.5, k)| for k=0..49 matches
  closed-form Γ-fn ratio
- 3 functions × steady-state replay determinism: same input → same
  output bytewise across two backtest runs

**FUTURE OPPORTUNITY:** Same pattern works for `volume_buf[]`. v5.16+
could add `ML_Compute_FracDiffVolume_d05` etc. as additional features
if the model finds price-frac-diff useful and we want to test volume-
