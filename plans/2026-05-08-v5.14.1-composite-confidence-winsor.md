// Composite confidence components (each [0, 1]).

struct RollingFreshness {
    uint64_t last_predict_us;   // wall-clock at last prediction
    double   tau_secs;           // decay time constant (cfg-tunable)
    // freshness = exp(-(now - last_predict) / (tau_secs * 1e6))
};

struct RollingCapacity {
    double current_adv;          // recent ADV estimate (rolling avg)
    double target_dollars;       // cfg-tunable; 0 = unbounded
    // capacity = (target_dollars > 0)
    //   ? min(1.0, kappa * current_adv / target_dollars) : 1.0

### Step 5 — FeatureStandardizer winsorization (.D)

File: `ML_Headers/FeatureStandardizer.hpp`

Extend sidecar binary format:
```cpp
struct FeatureStandardizer {
    // existing fields (mean[], std[], num_features, etc.)
    float winsor_low[NUM_FEATURES];   // v5.14.1.D NEW
    float winsor_high[NUM_FEATURES];  // v5.14.1.D NEW
    uint8_t has_winsor_bounds;        // v5.14.1.D NEW; 0=legacy=no clip
};
```

Apply path:
```cpp
inline int FeatureStandardizer_Apply(const FeatureStandardizer *s,
                                       float *features, int n) {
    // existing scaler apply (mean-center + unit-var)
    for (int i = 0; i < n; ++i) {
        if (s->has_winsor_bounds) {
            // 2 branchless cmovs per feature
            features[i] = std::fmin(std::fmax(features[i],
                                                s->winsor_low[i]),
                                       s->winsor_high[i]);
        }
        features[i] = (features[i] - s->mean[i]) / s->std[i];
    }
    // ... existing post-checks ...
}
```

**LATENCY:** ~5-15ns per inference (2 cmovs × N features); slow-path
cost; well within budget.
