def is_exempt(source_text: str, match_line_num: int) -> Optional[str]:
    """Return exemption rationale if match is exempted; None otherwise."""
    lines = source_text.splitlines()
    # Check match line + 3 lines above for exemption marker
    for i in range(max(0, match_line_num - 3), min(len(lines), match_line_num + 1)):

    """Verify TECH_DEBT-N has entry in closed.md. Sister verifier: verify_tech_debt_open."""
    n = match.group(1)
    closed_md = WORKSPACE_ROOT / "DOCS/tech-debt/closed.md"
    if not closed_md.exists():
        return None
    if re.search(rf'^id: TECH_DEBT-{n}\b', closed_md.read_text(), re.MULTILINE):
        return None  # CLEAN
    open_md = WORKSPACE_ROOT / "DOCS/tech-debt/open.md"
    if open_md.exists() and re.search(rf'^id: TECH_DEBT-{n}\b', open_md.read_text(), re.MULTILINE):
        return Finding('HIGH', f'TECH_DEBT-{n} NEW+CLOSED', source_file, source_line,
                       (n,), 'closed.md (currently in open.md)', 'open.md',
                       f'Move TECH_DEBT-{n} from open.md to closed.md OR re-classify claim',
                       promised_vs_landed)
    return Finding('HIGH', f'TECH_DEBT-{n} NEW+CLOSED', source_file, source_line,
                   (n,), 'closed.md', None,
                   f'Write TECH_DEBT-{n} entry to closed.md OR re-classify claim',
                   promised_vs_landed)


def verify_class_recurrence_bump(match, source_file, source_line, promised_vs_landed):
    """Verify Class N catalog frontmatter recurrence_count matches claimed bump (X→Y).
    
    Load-bearing verifier per Agent 2 HIGH-1: catches the exact pattern that caused the
    .B.7 Class 26 forward-promise drift (recurrence 11→13 promised DOCUMENTED-RISK entry
    that was never written; surfaced 2 ships later).
    """
    class_n = match.group(2)  # the digit captured from "Class N"
    bump_from = match.group(3)
    bump_to = match.group(4)
    # Find class-N catalog file
    catalog_glob = list((WORKSPACE_ROOT / "DOCS/recurring-bug-patterns").glob(f"class-{class_n}-*.md"))
    if not catalog_glob:
        return Finding('HIGH', f'Class {class_n} recurrence bump {bump_from}→{bump_to}',
                       source_file, source_line, (class_n, bump_from, bump_to),
                       f'DOCS/recurring-bug-patterns/class-{class_n}-*.md', None,
                       f'Class {class_n} catalog file not found; verify class number',
                       promised_vs_landed)
    catalog_text = catalog_glob[0].read_text()
    # Check frontmatter recurrence_count: <Y>
    rc_match = re.search(r'^recurrence_count:\s*(\d+)', catalog_text, re.MULTILINE)
    if rc_match is None:
        return Finding('HIGH', f'Class {class_n} recurrence bump claim',
                       source_file, source_line, (class_n, bump_from, bump_to),
                       f'{catalog_glob[0].name} frontmatter recurrence_count: {bump_to}',
                       'frontmatter missing recurrence_count field',
                       f'Add recurrence_count: {bump_to} to {catalog_glob[0].name} frontmatter',
                       promised_vs_landed)
    actual = int(rc_match.group(1))
    if actual != int(bump_to):
        return Finding('HIGH', f'Class {class_n} recurrence bump {bump_from}→{bump_to}',
                       source_file, source_line, (class_n, bump_from, bump_to),
                       f'recurrence_count: {bump_to}', f'recurrence_count: {actual}',
                       f'Update {catalog_glob[0].name} frontmatter recurrence_count: {actual}→{bump_to}',
                       promised_vs_landed)
    return None  # CLEAN


def verify_class_worked_example_added(match, source_file, source_line, promised_vs_landed):
    """Verify Class N catalog has NEW Worked Example for cited ship-tag."""
    class_n = match.group(1)
    ship_tag = match.group(2) if match.lastindex >= 2 and match.group(2) else None
    catalog_glob = list((WORKSPACE_ROOT / "DOCS/recurring-bug-patterns").glob(f"class-{class_n}-*.md"))
    if not catalog_glob:

                       source_file, source_line, (class_n, ship_tag),
                       f'{catalog_glob[0].name} Worked Examples section mentions .{ship_tag}',
                       'No mention of ship-tag in catalog body',
                       f'Add Worked Example entry for .{ship_tag} to {catalog_glob[0].name}',
                       promised_vs_landed)
    return None
