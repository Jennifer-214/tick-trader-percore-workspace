## § DriftOverride — bit layout diagram + accessors

### 8-byte struct layout

```
byte 0          byte 1     byte 2-3        byte 4-7
[flags]         [eps_idx]  [_padding1]     [_padding2]
 8 bits          8 bits     16 bits         32 bits

flags bit layout:
  bit 0  has_override      0=use defaults; 1=use this sidecar row
  bit 1  severity          0=REFUSE_STRICT (default); 1=WARN_ALWAYS
  bit 2  category          0=INFERENCE_CFG (default); 1=CROSS_BINARY
  bits 3-4  compare_kind   00=EPS_DEFAULT; 01=EPS_CUSTOM; 10=EPS_EXACT; 11=EPS_STRING
  bits 5-7  reserved       (3 bits headroom for future override semantics)

eps_idx byte:
  index into sparse g_drift_custom_eps[] table; only meaningful when compare_kind == EPS_CUSTOM
```

### Full struct definition

```cpp
// CoreFrameworks/CfgFieldDriftOverride.hpp
//
// Pattern: sidecar-override-pattern-for-registry-auto-flows.md Stage 3 ACTIVE (first canonical)
// Composes: bitmap-flag-api.md + multi-bit-state-encoding-pattern.md (canonical #6)
//
// v5.15.5.F.4d.1.C — NEW.

#pragma once

#include <cstdint>
#include <cstddef>

// Severity tokens (1 bit):
#define DRIFT_SEVERITY_REFUSE_STRICT  0u
#define DRIFT_SEVERITY_WARN_ALWAYS    1u

// Category tokens (1 bit):
#define DRIFT_CATEGORY_INFERENCE_CFG  0u
#define DRIFT_CATEGORY_CROSS_BINARY   1u

// Compare-kind tokens (2 bits):
#define DRIFT_COMPARE_EPS_DEFAULT     0u
#define DRIFT_COMPARE_EPS_CUSTOM      1u
#define DRIFT_COMPARE_EPS_EXACT       2u
#define DRIFT_COMPARE_EPS_STRING      3u

// Bit positions within flags byte:
inline constexpr uint8_t DRIFT_OVR_HAS_BIT          = 0;
inline constexpr uint8_t DRIFT_OVR_SEVERITY_BIT     = 1;
inline constexpr uint8_t DRIFT_OVR_CATEGORY_BIT     = 2;
inline constexpr uint8_t DRIFT_OVR_COMPARE_SHIFT    = 3;
inline constexpr uint8_t DRIFT_OVR_COMPARE_MASK     = 0x18; // bits 3-4 = 0b11000

struct DriftOverride {
    uint8_t flags;       // packed per layout above
    uint8_t eps_idx;     // index into g_drift_custom_eps[]
    int16_t _padding1 = 0;  // explicit per H12
    int32_t _padding2 = 0;  // explicit per H12
};
static_assert(sizeof(DriftOverride) == 8,
              "DriftOverride must be 8 bytes (cache-line packs 8 entries per 64B line)");

// Branchless accessors (multi-bit-state-encoding-pattern.md API style):
inline bool    drift_ovr_has(uint8_t flags)          { return  flags        & 0x01; }
inline uint8_t drift_ovr_severity(uint8_t flags)     { return (flags >> 1)  & 0x01; }
inline uint8_t drift_ovr_category(uint8_t flags)     { return (flags >> 2)  & 0x01; }
inline uint8_t drift_ovr_compare_kind(uint8_t flags) { return (flags & DRIFT_OVR_COMPARE_MASK) >> DRIFT_OVR_COMPARE_SHIFT; }

// Sparse eps values table (indexed by eps_idx when compare_kind == EPS_CUSTOM):
inline constexpr double g_drift_custom_eps[] = {
    /* [0] */ 1e-9,    // EPS_DEFAULT fallback (also used when compare_kind == EPS_DEFAULT)
    /* [1] */ 1e-3,    // xgb_subsample / xgb_colsample_bytree (XGBoost subsample fraction tolerance)
    /* [2] */ 1e-12,   // future tight-precision fields
    /* future: add more as cohorts grow */
};
```

### Auto-populate g_global_drift_overrides + g_per_core_drift_overrides arrays

```cpp
// Dispatch by which scope each field belongs to (per FIELD_IDX scheme).
// Note: all 5 XGBoost fields are GLOBAL at .C (training-time cfg);
// per-core override sidecar stays {0} until a per-core override field emerges.

#define EMIT_DRIFT_OVERRIDE_INIT(name, severity, category, compare, eps) \
    [FIELD_IDX_##name] = {                                               \
        .flags = (uint8_t)(                                              \
            (1u << DRIFT_OVR_HAS_BIT)                                    \
          | ((DRIFT_SEVERITY_##severity)  << DRIFT_OVR_SEVERITY_BIT)     \
          | ((DRIFT_CATEGORY_##category)  << DRIFT_OVR_CATEGORY_BIT)     \
          | ((DRIFT_COMPARE_##compare)    << DRIFT_OVR_COMPARE_SHIFT)    \
        ),                                                                \
        .eps_idx = (uint8_t)(eps),                                       \
    },

inline DriftOverride g_global_drift_overrides[FIELD_IDX_GLOBAL_END] = {
    FOREACH_DRIFT_OVERRIDE(EMIT_DRIFT_OVERRIDE_INIT)
};

inline DriftOverride g_per_core_drift_overrides[FIELD_IDX_PER_CORE_END] = {0};

#undef EMIT_DRIFT_OVERRIDE_INIT
```

## § Branchless sidecar lookup in `CFG_DRIFT_AUTOPOPULATE`

### Mask-select pattern for sidecar-vs-default dispatch

```cpp
// Extension of .B's emit_drift_check_per_row callback:
// Adds branchless sidecar lookup before applying per-cohort gate_when.

template <unsigned F>
inline void emit_drift_check_per_row(
    size_t idx,
    const CfgFieldDescriptor& d,
    void* ctxp)
{
    auto* ctx = static_cast<DriftCtx<F>*>(ctxp);

    // Branchless sidecar lookup (per-core vs global selected via descriptor flag):
    bool is_per_core = (d.lives_in_struct == CfgFieldDescriptor::PER_CORE_CFG);
    const DriftOverride& ovr =
        is_per_core ? g_per_core_drift_overrides[idx]
                     : g_global_drift_overrides[idx];

    // Mask-select: override_mask = all-1 when has_override; all-0 when not
    uint8_t override_mask = -(uint8_t)drift_ovr_has(ovr.flags);

    uint8_t severity = (drift_ovr_severity(ovr.flags) & override_mask)
                     | (DRIFT_SEVERITY_REFUSE_STRICT & ~override_mask);

    uint8_t category = (drift_ovr_category(ovr.flags) & override_mask)
                     | (DRIFT_CATEGORY_INFERENCE_CFG & ~override_mask);

    uint8_t compare_kind = (drift_ovr_compare_kind(ovr.flags) & override_mask)
                         | (DRIFT_COMPARE_EPS_DEFAULT & ~override_mask);

    // ... resolve cohort gate_when (per .B Step 11) ...
    // ... apply per-row drift compare using severity + category + compare_kind ...

    bool drifted = tt::cfg_drift_compare_descriptor<F>(
        d, *ctx->handle, *ctx->cfg, compare_kind, ovr.eps_idx);

    if (drifted) {
        if (severity == DRIFT_SEVERITY_WARN_ALWAYS) {
            log_warning("cfg drift on %s (sidecar WARN_ALWAYS; category=%s)",
                        d.cfg_field_name,
                        category == DRIFT_CATEGORY_CROSS_BINARY ? "cross_binary" : "inference_cfg");
        } else {
            BITMAP_SET(*ctx->failure_flags, FAILURE_MASK_cfg_inference_drift);
            (*ctx->drift_count)++;
        }
    }
}
```

### Design decision: severity branch is acceptable per H20 decision matrix
The `severity == WARN_ALWAYS` branch is at drift-check time (slow path; model-load boot-equivalent). Per H20 decision matrix: slow-path branches with cheap both-sides are acceptable. Mask-select for the value selection IS branchless; only the eventual log/fail action differs by severity.

## § RegistryRosterEntry — bit layout (extends `.A`)

### 40-byte struct layout

```
offset 0   8   16   24   25   26   27   28   32   40
       ptr ptr u8   u8   u8   u8   ptr  ptr  ─── struct end
       name source_file
                parent_idx
                     flags
                          bug_class
                                _padding=0
                                    design_spec
                                          doc

flags bit layout (uint8_t):
  bits 0-3  LEVEL              0=concrete; 1=cohort-meta; 2=root-meta (0-15 capacity)
  bits 4-5  WIRE_FORMAT_KIND   00=NOT_WIRE; 01=WIRE_FORMAT; 10=WIRE_FORMAT_TWO_SOURCE; 11=MIXED
  bits 6-7  reserved           (future expansion)

bug_class byte:
  Class N number (0-31 fits in 5 bits)
  bits 5-7 reserved (3 bits headroom for Class 32+)
```

### Full struct definition (from `.A` + `.C` verification)

```cpp
// CoreFrameworks/RegistryRoster.hpp (extends from .A)
//
// Pattern: meta-registry-pattern-for-codebase-registry-discipline.md
// Composes: bitmap-flag-api.md + multi-bit-state-encoding-pattern.md (canonical #7)

struct RegistryRosterEntry {
    const char* name;
    const char* source_file;
    uint8_t     parent_idx;     // 0xFF = ROOT; else index into g_registry_roster[]
    uint8_t     flags;          // PACKED: bits 0-3 LEVEL; bits 4-5 WIRE_FORMAT_KIND; bits 6-7 reserved
    uint8_t     bug_class;      // 0 = NONE; else Class N (5 bits used; 3 bits reserved)
    uint8_t     _padding = 0;   // explicit per H12
    const char* design_spec;
    const char* doc;
};
static_assert(sizeof(RegistryRosterEntry) == 40,
              "RegistryRosterEntry must be 40 bytes (2 entries per cache line)");

// Branchless accessors:
inline constexpr uint8_t ROSTER_FLAGS_LEVEL_MASK     = 0x0F;
inline constexpr uint8_t ROSTER_FLAGS_WIRE_KIND_MASK = 0x30;
inline constexpr uint8_t ROSTER_FLAGS_WIRE_KIND_SHIFT = 4;

inline uint8_t roster_level(uint8_t flags)            { return flags & ROSTER_FLAGS_LEVEL_MASK; }
inline uint8_t roster_wire_format_kind(uint8_t flags) { return (flags & ROSTER_FLAGS_WIRE_KIND_MASK) >> ROSTER_FLAGS_WIRE_KIND_SHIFT; }
```

## § ManualFieldInventoryEntry — bit layout (NEW at `.C`)

### 24-byte struct layout

```
offset 0   8    16   17   18   20   24
       ptr ptr  u8   u8   i16  ptr  ─── struct end
       struct_name field_name
                kind
                     rationale_idx
                          _padding=0
                              migration_trigger

kind byte:
  bits 0-2  CATEGORY     0=DERIVED; 1=RUNTIME_STATE; 2=PER_CORE_OVERRIDE; 3=INTERNAL; 4-7 future
  bits 3-7  reserved     (5 bits headroom)
```

### Full struct definition

```cpp
// MemHeaders/ManualFieldInventory.hpp (NEW at .C)
//
// Pattern: manual-fields-inventory-pattern.md Stage 2 DRAFT
// Composes: multi-bit-state-encoding-pattern.md (canonical #8)
//
// v5.15.5.F.4d.1.C — NEW.

#pragma once

#include <cstdint>

// Category tokens (3 bits):
#define MFI_CATEGORY_DERIVED            0u  // computed from other fields
#define MFI_CATEGORY_RUNTIME_STATE      1u  // mutable runtime state (not cfg)
#define MFI_CATEGORY_PER_CORE_OVERRIDE  2u  // per-core override surface
#define MFI_CATEGORY_INTERNAL           3u  // engine internal (not operator-facing)

// Bit positions:
inline constexpr uint8_t MFI_CATEGORY_SHIFT  = 0;
inline constexpr uint8_t MFI_CATEGORY_MASK   = 0x07;  // bits 0-2

struct ManualFieldInventoryEntry {
    const char* struct_name;       // e.g., "OrderManagerState"
    const char* field_name;        // e.g., "ezoo_refs"
    uint8_t     kind;               // PACKED per layout above
    uint8_t     rationale_idx;      // index into sparse rationale-strings table
    int16_t     _padding = 0;       // explicit per H12
    const char* migration_trigger;  // e.g., "promote to FOREACH_*_CFG_FIELD when ..."
};
static_assert(sizeof(ManualFieldInventoryEntry) == 24,
              "ManualFieldInventoryEntry must be 24 bytes");

// Branchless accessors:
inline uint8_t mfi_category(uint8_t kind) { return (kind & MFI_CATEGORY_MASK) >> MFI_CATEGORY_SHIFT; }

// Sparse rationale-strings table (indexed by rationale_idx):
inline constexpr const char* g_mfi_rationale_strings[] = {
    /* [0] */ "SPECIAL_CLEAR_HELPER — bit-packed clear semantics differ from registry reset",
    /* [1] */ "CROSS_LAYER_TYPE_ERASURE — void* field for cross-layer reference (see type-erased-per-core-resource-handle-pattern.md)",
    /* [2] */ "RUNTIME_DERIVED — computed at boot from cfg fields; not a cfg field itself",
    /* future: add more as exemptions grow */
};
```

### Design decision: LEVEL=0, PARENT=DERIVED_FILTER
FOREACH_DRIFT_OVERRIDE is a Level-0 concrete data registry (flat row enumeration; no sub-cohorts). Its PARENT is DERIVED_FILTER (Level-1 meta) because the sidecar consumes the derived filter's parent registry's flagged rows (each sidecar row maps to a FIELD_IDX of a STAMP_BOUND_CFG_DERIVED-flagged source row).
