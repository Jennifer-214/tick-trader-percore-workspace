if (cfg.maker_orders_enabled && book_snapshot_valid) {
    // Compute limit price based on side + offset

    if (side == ORDER_BUY) {
        FPN<F> tick = cfg.maker_tick_size;

### Step 4 — Drainer: cancel-and-replace stale-limit check

File: `CoreFrameworks/OrderManager.hpp` (drainer-thread sweep)

```cpp
// v5.14.7 — OMS_CheckStaleLimits called from OrderManager_Tick
// alongside existing fill-handling logic.
template <unsigned F>
inline void OMS_CheckStaleLimits(OrderManagerState<F>* oms,
                                   uint64_t now_us) {
    uint16_t bm = oms->order_bitmap;
    while (bm) {
        int idx = __builtin_ctz(bm);
        bm &= (uint16_t)(bm - 1);
        Order<F>* o = &oms->orders[idx];
        if (o->state != ORDER_SUBMITTED) continue;
        if (o->cancel_after_us == 0) continue;
        if (now_us < o->cancel_after_us) continue;
        // Stale; cancel + optional replace
        BinanceAdapter_CancelOrder(&oms->adapter, "BTCUSDT", o->id);
        if (cfg.maker_fallback_to_market) {
            // re-submit as MARKET via OMS_PushSubmit
            OMS_PushSubmit(oms, o->core_id,
                o->type == ORDER_LIMIT_BUY ? ORDER_MARKET_BUY
                                            : ORDER_MARKET_SELL,
                o->qty, o->intended_tp, o->intended_sl,
                o->strategy_id, o->event_price);
        }
    }
}
```
