```cpp
// MemHeaders/CfgDriftGate.hpp — Cohort gate dispatch sidecar
//
// Pattern: sidecar-override-pattern-for-registry-auto-flows.md Stage 4 (2nd canonical)
// + branchless-dispatch-discipline.md Pattern 1 (fn pointer table) per H20
// + multi-state-dispatch-with-per-state-update-metadata.md Stage 3 ACTIVE
//
// Per-concern separation discipline: cohort gate dispatch is SEPARATE from
// custom-semantics override dispatch (FOREACH_DRIFT_OVERRIDE at .C); each
// concern gets its own sidecar. Closes Class 18 mirror at the conflate-concerns
// layer.

enum DriftGateKind : uint8_t {
    DRIFT_GATE_DEFAULT              = 0,  // STAMP_HAS only (implicit; sidecar row absent)
    DRIFT_GATE_BANDIT_THOMPSON      = 1,
    DRIFT_GATE_RISK_DEGRADATION     = 2,
    DRIFT_GATE_RIDGE_ANY            = 3,
    DRIFT_GATE_COMPOSITE_CONFIDENCE = 4,
    // Winsor cohort is Default (cfg parse-time validation enforces bounds)
    DRIFT_GATE_KIND_COUNT
};

// Per-cohort gate fns (pure; testable in isolation):
template <unsigned F>
inline bool gate_default(const ControllerConfig<F>* /*cfg*/, const ModelStampResult* h)
    { return STAMP_HAS(*h, inference_cfg); }

template <unsigned F>
inline bool gate_bandit_thompson(const ControllerConfig<F>* cfg, const ModelStampResult* h)
    { return STAMP_HAS(*h, inference_cfg) && BITMAP_IS_SET(cfg->ml_cfg_flags, MASK_ML_CFG_BANDIT_ENABLED); }

template <unsigned F>
inline bool gate_risk_degradation(const ControllerConfig<F>* cfg, const ModelStampResult* h)
    { return STAMP_HAS(*h, inference_cfg) && cfg->risk_degradation_curve != 0; }

template <unsigned F>
inline bool gate_ridge_any(const ControllerConfig<F>* cfg, const ModelStampResult* h) {
    return STAMP_HAS(*h, inference_cfg)
        && BITMAP_ANY(cfg->ml_cfg_flags,
                       MASK_ML_CFG_RIDGE_WITHIN_HORIZON | MASK_ML_CFG_RIDGE_ACROSS_HORIZONS);
}

template <unsigned F>
inline bool gate_composite_confidence(const ControllerConfig<F>* cfg, const ModelStampResult* h)
    { return STAMP_HAS(*h, inference_cfg) && BITMAP_IS_SET(cfg->ml_cfg_flags, MASK_ML_CFG_CONFIDENCE_COMPOSITE_ENABLED); }

// Branchless dispatch table per H20 Pattern 1:
template <unsigned F>
using DriftGateFn = bool (*)(const ControllerConfig<F>* cfg, const ModelStampResult* h);

template <unsigned F>
inline constexpr DriftGateFn<F> g_drift_gate_table[DRIFT_GATE_KIND_COUNT] = {
    [DRIFT_GATE_DEFAULT]              = &gate_default<F>,
    [DRIFT_GATE_BANDIT_THOMPSON]      = &gate_bandit_thompson<F>,
    [DRIFT_GATE_RISK_DEGRADATION]     = &gate_risk_degradation<F>,
    [DRIFT_GATE_RIDGE_ANY]            = &gate_ridge_any<F>,
    [DRIFT_GATE_COMPOSITE_CONFIDENCE] = &gate_composite_confidence<F>,
};

// Sparse sidecar registry: per-FIELD_IDX cohort tag for non-Default rows.
// Default cohort (sidecar entry absent / value 0) is implicit for unflagged rows.
#define FOREACH_DRIFT_GATE(X)                                              \
    /* Bandit/Thompson cohort (4 fields) */                                \
    X(bandit_algorithm,                  BANDIT_THOMPSON)                  \
    X(thompson_mu_prior,                 BANDIT_THOMPSON)                  \
    X(thompson_precision_prior,          BANDIT_THOMPSON)                  \
    X(thompson_precision_obs,            BANDIT_THOMPSON)                  \
    /* Soft-risk degradation (3 fields; risk_degradation_curve gates self) */ \
    X(risk_full_size_threshold,          RISK_DEGRADATION)                 \
    X(risk_min_size_threshold,           RISK_DEGRADATION)                 \
    X(risk_min_size_pct,                 RISK_DEGRADATION)                 \
    /* Ridge cohort (3 scalars; ridge_within/across gates themselves) */   \
    X(ridge_lambda,                      RIDGE_ANY)                        \
    X(ridge_cost_penalty,                RIDGE_ANY)                        \
    X(ridge_min_ic_floor,                RIDGE_ANY)                        \
    /* Composite confidence cohort (4 scalars) */                          \
    X(confidence_freshness_tau_secs,     COMPOSITE_CONFIDENCE)             \
    X(confidence_capacity_target_dollars, COMPOSITE_CONFIDENCE)            \
    X(confidence_capacity_kappa,         COMPOSITE_CONFIDENCE)             \
    X(confidence_rmse_baseline,          COMPOSITE_CONFIDENCE)

// Dense sparse-arrays indexed by FIELD_IDX (default 0 = DRIFT_GATE_DEFAULT):
#define EMIT_DRIFT_GATE_INIT(name, kind) \
    [FIELD_IDX_##name] = DRIFT_GATE_##kind,

inline DriftGateKind g_drift_gate_lookup_per_core[FIELD_IDX_PER_CORE_END] = {
    FOREACH_DRIFT_GATE(EMIT_DRIFT_GATE_INIT)  // entries for per-core fields
};
inline DriftGateKind g_drift_gate_lookup_global[FIELD_IDX_GLOBAL_END] = {
    FOREACH_DRIFT_GATE(EMIT_DRIFT_GATE_INIT)  // entries for global fields
};
#undef EMIT_DRIFT_GATE_INIT
```

## Anti-patterns to avoid

### Class 18 (mirror-incomplete) — DON'T leave POST_CFG manual entries after migration
The whole point of `.B` is to ELIMINATE the mirror between `FOREACH_STAMP_BOUND_CFG` (manual) and `FOREACH_CFG_FIELD STAMP_BOUND_CFG_DERIVED` (derived). Leaving manual POST_CFG entries means BOTH paths emit the field → either double-emit (canonical body has 2 lines for the field) OR they go out of sync (Class 18 latent). Verify each step deletes its manual POST_CFG entries.

### Class 21 (parallel descriptors) — DON'T leave `FOREACH_STAMP_BOUND_CFG` with content
The legacy registry must be empty (or DELETED entirely) post-migration. Leaving rows there with the framework also walking the derived filter = parallel descriptors at the registry layer.

### Class 23 (type-erased reinterpret_cast) — DON'T introduce manual cast paths in CFG_DRIFT_AUTOPOPULATE
Use `tt::cfg_drift_compare<T>` for per-row comparison; don't `*reinterpret_cast<T*>((char*)cfg + offset)`. The walker callback already passes the descriptor + ctx; cast through typed dispatch.

### Class 14 (plan API drift) — Verify CFG_DRIFT_AUTOPOPULATE callback sig matches consumer expectations
At consumer migration sites (CoreModelZoo / StampHelper / ConfidenceScore), the existing `sr.inference_cfg_drift_count` semantics must be preserved. The new macro's callback should increment the count + set the failure flag identically.

### Wire-format byte preservation (per H9)
**`.B`'s migration must NOT change canonical body byte shape per row.** The framework walker should produce identical bytes for each field as the legacy `FOREACH_STAMP_BOUND_CFG` walker did. Invariants I1-I5 from `.A` verify this structurally; plus at `.D` the v5.14 fixture round-trip test (Layer 4) catches any byte drift.

### Locale-pin discipline (per Layer 2 + `.A`'s framework)
`tt::cfg_emit_synthetic_field<T>` MUST use locale-pinned snprintf. Inherited from framework macro's emit fn (which already pins per Layer 2). Verify by I3 invariant (no comma decimals in body even under simulated LC_NUMERIC=de_DE).

### Cohort gate_when expression discipline
Per Charter 12: always-emit canonical means EVERY flagged row emits to stamp. `gate_when` only affects drift-check skip/fire. Forgetting this distinction = silent drift-check skip on cohort-disabled fields (false negative).
